	# ------------------------------------------------------------
	# 河童工業化計畫
	#
	# timed decision 的流程：
	# 1. 點擊時支付 cost，modifier 在讀條期間生效。
	# 2. days_remove 結束後執行 remove_effect。
	# 3. 不要在 target_trigger 中排除點擊時才寫入的 active flag，
	#    否則決議會在啟動後立刻失去合法目標並消失。
	# ------------------------------------------------------------

	# 民用工業：+1 共用建築槽、+1 民用工廠
	GSK_indproj_civ_1 = {
		icon = GFX_decision_generic_build_civ_factory
		state_target = yes
		on_map_mode = map_and_decisions_view
		targets = { 282 1018 1019 1020 526 528 529 530 531 533 535 }

		cost = 25
		days_remove = 45
		fire_only_once = yes

		modifier = {
			civilian_factory_use = 1
		}

		target_trigger = {
			FROM = {
				is_owned_and_controlled_by = ROOT
				NOT = { has_state_flag = GSK_indproj_civ_1_done }
			}
		}

		available = {
			num_of_civilian_factories_available_for_projects > 0
		}

		remove_effect = {
			FROM = {
				set_state_flag = GSK_indproj_civ_1_done
				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}
	}

	# 特殊設施：282 陸軍設施、1082 空軍設施、530 核設施
	GSK_indproj_special_facility = {
		icon = GFX_decision_generic_construction
		state_target = yes
		on_map_mode = map_and_decisions_view
		targets = { 282 1082 530 }

		cost = 50
		days_remove = 120
		fire_only_once = yes

		modifier = {
			civilian_factory_use = 4
		}

		target_trigger = {
			FROM = {
				is_owned_and_controlled_by = ROOT
				NOT = { has_state_flag = GSK_indproj_special_facility_done }

				OR = {
					AND = {
						state = 282
						free_building_slots = {
							building = land_facility
							size > 0
							include_locked = yes
						}
					}
					AND = {
						state = 1082
						free_building_slots = {
							building = air_facility
							size > 0
							include_locked = yes
						}
					}
					AND = {
						state = 530
						free_building_slots = {
							building = nuclear_facility
							size > 0
							include_locked = yes
						}
					}
				}
			}
		}

		available = {
			num_of_civilian_factories_available_for_projects > 3
		}

		remove_effect = {
			FROM = {
				set_state_flag = GSK_indproj_special_facility_done

				if = {
					limit = { state = 282 }
					add_building_construction = {
						type = land_facility
						level = 1
						instant_build = yes
					}
				}
				if = {
					limit = { state = 1082 }
					add_building_construction = {
						type = air_facility
						level = 1
						instant_build = yes
					}
				}

						level = 1
						instant_build = yes
					}
				}
			}
		}
	}
