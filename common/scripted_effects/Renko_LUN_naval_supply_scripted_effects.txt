Renko_LUN_grant_dynamic_navy = {
	if = {
		limit = {
			exists = LUN
		}
		LUN = {
			if = {
				limit = {
					any_owned_state = {
						is_controlled_by = LUN
						naval_base > 0
					}
				}
				Renko_LUN_navy_grant_patrol_package = yes
				Renko_LUN_navy_grant_strike_package = yes
				Renko_LUN_navy_grant_submarine_package = yes
			}
		}
	}
}
