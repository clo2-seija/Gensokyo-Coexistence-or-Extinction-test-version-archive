KYO_start_lunarian_surface_war = {
	if = {
		limit = { NOT = { has_country_flag = KYO_lunarian_surface_war_started } }
		set_country_flag = KYO_lunarian_surface_war_started
		clr_country_flag = KYO_surface_liberation_completed
		set_variable = { KYO_surface_selected = 0 }
		set_variable = { KYO_surface_liberated = 0 }
		clr_country_flag = KYO_surface_USA_selected
		clr_country_flag = KYO_surface_USA_capital_breached
		clr_country_flag = KYO_surface_USA_liberated
		clr_country_flag = KYO_surface_anchor_USA
		if = {
			limit = { country_exists = USA USA = { is_major = yes has_capitulated = no NOT = { is_subject_of = KYO } } }
			set_country_flag = KYO_surface_USA_selected
			add_to_variable = { KYO_surface_selected = 1 }
			USA = {
				if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
				else_if = { limit = { is_in_faction = yes } leave_faction = yes }
				remove_ideas = KYO_lunarian_crossline_disrupted
				remove_ideas = KYO_lunarian_defeat_pressure_1
				remove_ideas = KYO_lunarian_defeat_pressure_2
				remove_ideas = KYO_lunarian_defeat_pressure_3
				remove_ideas = KYO_lunarian_faction_lock
				set_country_flag = KYO_lunarian_controlled_major
			}
		}
		clr_country_flag = KYO_surface_ENG_selected
		clr_country_flag = KYO_surface_ENG_capital_breached
		clr_country_flag = KYO_surface_ENG_liberated
		clr_country_flag = KYO_surface_anchor_ENG

				if = { limit = { is_faction_leader = yes } dismantle_faction = yes }

		}

# A lunar-series defeat only breaks control over the United States.  The first
# surviving controlled major becomes the new faction anchor, and the liberated
# United States enters war against every country that remains controlled.
KYO_release_lunarian_controlled_usa = {
	if = {
		limit = {
			has_country_flag = KYO_surface_USA_selected
			NOT = { has_country_flag = KYO_surface_USA_liberated }
			country_exists = USA
			USA = { has_country_flag = KYO_lunarian_controlled_major }
		}
		set_country_flag = KYO_surface_USA_liberated
		add_to_variable = { KYO_surface_liberated = 1 }
		clr_country_flag = KYO_surface_anchor_USA
		clr_country_flag = KYO_surface_anchor_ENG
		clr_country_flag = KYO_surface_anchor_GER
		clr_country_flag = KYO_surface_anchor_SOV
		clr_country_flag = KYO_surface_anchor_FRA
		USA = {
			if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
			else_if = { limit = { is_in_faction = yes } leave_faction = yes }
			every_enemy_country = { white_peace = PREV }
			clr_country_flag = KYO_lunarian_controlled_major
			KYO_refresh_lunarian_control_modifier = yes
		}
		# Temporarily remove the faction-locking tier ideas while the controlled
		# faction is rebuilt around a surviving member.
		if = { limit = { country_exists = ENG } ENG = { KYO_remove_lunarian_control_ideas = yes } }
		if = { limit = { country_exists = GER } GER = { KYO_remove_lunarian_control_ideas = yes } }
		if = { limit = { country_exists = SOV } SOV = { KYO_remove_lunarian_control_ideas = yes } }
		if = { limit = { country_exists = FRA } FRA = { KYO_remove_lunarian_control_ideas = yes } }
		if = {
			limit = {
				has_country_flag = KYO_surface_ENG_selected
				NOT = { has_country_flag = KYO_surface_ENG_liberated }
				country_exists = ENG
				ENG = { has_country_flag = KYO_lunarian_controlled_major }
			}
			set_country_flag = KYO_surface_anchor_ENG
			ENG = {
				if = {
					limit = { is_faction_leader = yes }
					set_faction_name = KYO_lunarian_surface_front
				}
				else = {
					if = { limit = { is_in_faction = yes } leave_faction = yes }
					create_faction_from_template = faction_template_generic_dominance
				set_faction_name =  KYO_lunarian_surface_front
				}
			}
		}
		else_if = {
			limit = {
				has_country_flag = KYO_surface_GER_selected
				NOT = { has_country_flag = KYO_surface_GER_liberated }
				country_exists = GER
				GER = { has_country_flag = KYO_lunarian_controlled_major }
			}
			set_country_flag = KYO_surface_anchor_GER
			GER = {
				if = {
					limit = { is_faction_leader = yes }
					set_faction_name = KYO_lunarian_surface_front
				}
				else = {
					if = { limit = { is_in_faction = yes } leave_faction = yes }
					create_faction = KYO_lunarian_surface_front
				}
			}
		}
		else_if = {
			limit = {
				has_country_flag = KYO_surface_SOV_selected
				NOT = { has_country_flag = KYO_surface_SOV_liberated }
				country_exists = SOV
				SOV = { has_country_flag = KYO_lunarian_controlled_major }
			}
			set_country_flag = KYO_surface_anchor_SOV
			SOV = {
				if = {
					limit = { is_faction_leader = yes }
					set_faction_name = KYO_lunarian_surface_front
				}
				else = {
					if = { limit = { is_in_faction = yes } leave_faction = yes }
					create_faction_from_template = faction_template_generic_dominance
				set_faction_name =  KYO_lunarian_surface_front
				}
			}
		}
		else_if = {
			limit = {
				has_country_flag = KYO_surface_FRA_selected
				NOT = { has_country_flag = KYO_surface_FRA_liberated }
				country_exists = FRA
				FRA = { has_country_flag = KYO_lunarian_controlled_major }
			}
			set_country_flag = KYO_surface_anchor_FRA
			FRA = {
				if = {
					limit = { is_faction_leader = yes }

					if = { limit = { is_in_faction = yes } leave_faction = yes }
					create_faction_from_template = faction_template_generic_dominance
				set_faction_name =  KYO_lunarian_surface_front
				}
			}
		}
		KYO_enforce_lunarian_surface_faction = yes
		KYO_refresh_all_lunarian_control_modifiers = yes
		USA = {
			if = {
				limit = {
					country_exists = ENG
					ENG = { has_country_flag = KYO_lunarian_controlled_major }
					NOT = { has_war_with = ENG }
					NOT = { is_in_faction_with = ENG }
				}
				declare_war_on = { target = ENG type = annex_everything }
			}
			if = {
				limit = {
					country_exists = GER
					GER = { has_country_flag = KYO_lunarian_controlled_major }
					NOT = { has_war_with = GER }
					NOT = { is_in_faction_with = GER }
				}
				declare_war_on = { target = GER type = annex_everything }
			}

					country_exists = SOV

					NOT = { has_war_with = SOV }
					NOT = { is_in_faction_with = SOV }

			}
			if = {

					NOT = { has_war_with = FRA }
					NOT = { is_in_faction_with = FRA }

KYO_lunar_resolve_round = {
	set_variable = { KYO_lunar_last_X = KYO_lunar_X }
	set_variable = { KYO_lunar_last_T = KYO_lunar_T }
	if = {
		limit = { check_variable = { KYO_lunar_X > KYO_lunar_T } }
		add_to_variable = { KYO_lunar_attacker_wins = 1 }
		set_country_flag = KYO_lunar_last_round_attacker
		clr_country_flag = KYO_lunar_last_round_defender
		KYO_disrupt_lunarian_surface_front = yes
		KYO_apply_lunarian_defeat_pressure = yes
		KYO_apply_lunar_victory_milestones = yes
		if = { limit = { check_variable = { KYO_lunar_attacker_wins > 4 } check_variable = { KYO_lunar_attacker_wins < 6 } } country_event = { id = yakumo.67 hours = 6 } }
	}
	else = {
		add_to_variable = { KYO_lunar_defender_wins = 1 }
		set_country_flag = KYO_lunar_last_round_defender
		clr_country_flag = KYO_lunar_last_round_attacker
		if = { limit = { NOT = { has_country_flag = KYO_lunar_first_loss_reported } } set_country_flag = KYO_lunar_first_loss_reported country_event = { id = yakumo.65 hours = 6 } }
	}
	set_variable = { KYO_lunar_score_lead = KYO_lunar_attacker_wins }
	subtract_from_variable = { KYO_lunar_score_lead = KYO_lunar_defender_wins }
	if = {
		limit = { check_variable = { KYO_lunar_attacker_wins > 5 } }
		set_country_flag = KYO_lunar_campaign_won
		set_country_flag = KYO_lunar_crossline_reinforcement_disabled
		KYO_degrade_lunarian_surface_control = yes
		clr_country_flag = KYO_lunar_campaign_active
		country_event = { id = yakumo.68 hours = 6 }
	}
	else_if = {
		limit = { check_variable = { KYO_lunar_defender_wins > 5 } }
		clr_country_flag = KYO_lunar_campaign_active
		country_event = { id = yakumo.74 hours = 6 }
	}
	else = { add_to_variable = { KYO_lunar_round = 1 } KYO_lunar_start_round = yes }
}
