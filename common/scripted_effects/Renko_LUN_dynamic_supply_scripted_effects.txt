Renko_LUN_supply_infantry_equipment = {
	if = {
		limit = {
			OR = {
				AND = {
					has_tech = infantry_weapons
				}
				AND = {
					has_tech = infantry_weapons1
				}
				AND = {
					has_tech = improved_infantry_weapons
				}
				AND = {
					has_tech = advanced_infantry_weapons
				}
			}
		}
		meta_effect = {
			text = {
				add_equipment_to_stockpile = {
					type = infantry_equipment_[Renko_LUN_supply_tier]
					amount = $Renko_AMOUNT$
					producer = LUN
				}
			}
			Renko_LUN_supply_tier = "[GetRenko_LUN_supply_infantry_equipment]"
		}
	}
}

Renko_LUN_grant_dynamic_equipment = {
	if = {
		limit = {
			exists = LUN
		}
		LUN = {
			if = {
				limit = {
					capital_scope = {
						is_controlled_by = LUN
					}
				}
				if = {
					limit = {
						NOT = {
							has_template = "Renko Lunar Infantry Division"
						}
					}
					Renko_LUN_create_infantry_template = yes
				}
				capital_scope = {
					create_unit = {
						division = "division_template = \"Renko Lunar Infantry Division\" start_experience_factor = 0.30 start_equipment_factor = 1.0"
						owner = PREV
						count = 4
					}
				}
				Renko_LUN_supply_infantry_reserve = yes
				if = {
					limit = {
						NOT = {
							has_template = "Renko Lunar Armored Division"
						}
					}
					Renko_LUN_create_armored_template = yes
				}
				capital_scope = {
					create_unit = {
						division = "division_template = \"Renko Lunar Armored Division\" start_experience_factor = 0.40 start_equipment_factor = 1.0"
						owner = PREV
						count = 4
					}
				}
				Renko_LUN_supply_armored_reserve = yes
				Renko_LUN_supply_air_reserve = yes
			}
		}
	}
}
