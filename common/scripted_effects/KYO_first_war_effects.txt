# KYO 第一次人妖大戰 v1.0 — 共用效果
# 玩家實際獲得的效果保持可見；hidden_effect 僅包 clamp / refresh / check 等內部維護呼叫。
# 玩家會直接看到的入口效果使用 custom_effect_tooltip；實際加成放在 hidden_effect。

KYO_refresh_military_committee_idea = {
	if = {
		limit = {
			has_idea = KYO_military_committee_fulfillment_0
		}
		remove_ideas = KYO_military_committee_fulfillment_0
	}
	if = {
		limit = {
			has_idea = KYO_military_committee_fulfillment_1
		}
		remove_ideas = KYO_military_committee_fulfillment_1
	}
	if = {
		limit = {
			has_idea = KYO_military_committee_fulfillment_2
		}
		remove_ideas = KYO_military_committee_fulfillment_2
	}
	if = {
		limit = {
			has_idea = KYO_military_committee_fulfillment_3
		}
		remove_ideas = KYO_military_committee_fulfillment_3
	}
	if = {
		limit = {
			check_variable = {
				KYO_fulfillment_military_committee > 74
			}
		}
		add_ideas = KYO_military_committee_fulfillment_3
	}
	else_if = {
		limit = {
			check_variable = {
				KYO_fulfillment_military_committee > 49
			}
		}
		add_ideas = KYO_military_committee_fulfillment_2
	}
	else_if = {
		limit = {
			check_variable = {
				KYO_fulfillment_military_committee > 24
			}
		}
		add_ideas = KYO_military_committee_fulfillment_1
	}
	else = {
		add_ideas = KYO_military_committee_fulfillment_0
	}
}

KYO_clamp_and_refresh_fulfillment = {
	if = {
		limit = {
			check_variable = {
				KYO_fulfillment_military_committee > 100
			}
		}
		set_variable = {
			KYO_fulfillment_military_committee = 100
		}
	}
	if = {
		limit = {
			check_variable = {
				KYO_fulfillment_military_committee < 0
			}
		}
		set_variable = {
			KYO_fulfillment_military_committee = 0
		}
