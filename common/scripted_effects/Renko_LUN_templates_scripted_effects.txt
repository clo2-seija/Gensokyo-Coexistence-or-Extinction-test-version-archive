	set_technology = {
		Renko_LUN_invasion_units_tech = 1
		infantry_weapons = 1
		infantry_weapons1 = 1
		support_weapons = 1
		tech_support = 1
		tech_engineers = 1
		gw_artillery = 1
		tech_military_police = 1
		tech_trucks = 1
		gwtank_chassis = 1
		basic_medium_tank_chassis = 1
		basic_heavy_tank_chassis = 1
		interwar_antiair = 1
		interwar_antitank = 1
		popup = no
	}
	if = {
		limit = { NOT = { is_special_project_completed = sp:sp_air_helicopter } }
		complete_special_project = sp:sp_air_helicopter
	}
	if = {
		limit = { NOT = { is_special_project_completed = sp:sp_land_military_engineering_vehicles } }
		complete_special_project = sp:sp_land_military_engineering_vehicles
	}
	if = {
		limit = { NOT = { is_special_project_completed = sp:sp_land_flamethrower_tank } }
		complete_special_project = sp:sp_land_flamethrower_tank
	}
	set_technology = {
		sp_armored_engineer_tech = 1
		sp_helicopter_transport_pods_tech = 1
		sp_helicopter_artillery_observers_tech = 1
		sp_helicopter_med_evac_tech = 1
		popup = no
	}
}

Renko_LUN_grant_army_templates_and_units = {
	Renko_LUN_unlock_template_requirements = yes
	if = {
		limit = { NOT = { has_template = "Renko Lunar Infantry Division" } }
		Renko_LUN_create_infantry_template = yes
	}
	if = {
		limit = { NOT = { has_template = "Renko Lunar Armored Division" } }
		Renko_LUN_create_armored_template = yes
	}
	if = {
		limit = { NOT = { has_template = "Renko Lunar Cavalry Garrison" } }
		Renko_LUN_create_suppression_template = yes
	}
	capital_scope = {
		create_unit = {
			division = "division_template = \"Renko Lunar Infantry Division\" start_equipment_factor = 1.0"
			owner = PREV
			count = 100
		}
		create_unit = {
			division = "division_template = \"Renko Lunar Armored Division\" start_equipment_factor = 1.0"
			owner = PREV
			count = 20
		}
	}
}
