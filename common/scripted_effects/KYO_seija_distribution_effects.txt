KYO_seija_distribution_init = {
	if = {
		limit = { NOT = { has_country_flag = KYO_seija_distribution_unlocked } }
		set_country_flag = KYO_seija_distribution_unlocked
		set_variable = { KYO_seija_supplies_total = 5 }
		set_variable = { KYO_common_youkai_supply = 1 }
		set_variable = { KYO_human_supply = 1 }
		set_variable = { KYO_revolutionary_supply = 1 }
		set_variable = { KYO_opposition_supply = 1 }
		add_ideas = KYO_reconstruction_command
	}
	hidden_effect = {
		# Always resynchronise the derived counter, including existing saves.
		KYO_seija_distribution_refresh_all = yes
	}
}

# KYO_seija_supplies_allocated is derived data. Rebuild it from the four
# allocation variables so it can never desynchronise from the real values.
KYO_seija_recalculate_allocated_supply = {
	set_variable = {
		KYO_seija_supplies_allocated = {
			value = KYO_common_youkai_supply
			add = KYO_human_supply
			add = KYO_revolutionary_supply
			add = KYO_opposition_supply
		}
	}
}

KYO_seija_refresh_common_youkai_idea = {
	if = {
		limit = { check_variable = { KYO_common_youkai_supply > 4 } }
		add_ideas = KYO_common_youkai_satisfaction_5
	}
	else_if = {
		limit = { check_variable = { KYO_common_youkai_supply > 3 } }
		add_ideas = KYO_common_youkai_satisfaction_4
	}
	else_if = {
		limit = { check_variable = { KYO_common_youkai_supply > 2 } }
		add_ideas = KYO_common_youkai_satisfaction_3
	}
	else_if = {
		limit = { check_variable = { KYO_common_youkai_supply > 1 } }
		add_ideas = KYO_common_youkai_satisfaction_2
	}
	else_if = {
		limit = { check_variable = { KYO_common_youkai_supply > 0 } }
		add_ideas = KYO_common_youkai_satisfaction_1
	}
	else = {
		add_ideas = KYO_common_youkai_satisfaction_0
	}
}

KYO_seija_common_youkai_add_supply = {
	if = {
		limit = {
			KYO_seija_has_free_supply = yes
			check_variable = { KYO_common_youkai_supply < 5 }
		}
		add_to_variable = { KYO_common_youkai_supply = 1 }
		KYO_seija_distribution_refresh_all = yes
	}
}

KYO_seija_monthly_bad_event_check = {
	if = {
		limit = {
			has_country_flag = KYO_seija_distribution_unlocked
			OR = {
				check_variable = { KYO_common_youkai_supply < 2 }
				check_variable = { KYO_human_supply < 2 }
				check_variable = { KYO_revolutionary_supply < 2 }
				check_variable = { KYO_opposition_supply < 2 }
			}
		}
		random_list = {
			35 = { country_event = kyo_seija_distribution.20 }
			35 = { country_event = kyo_seija_distribution.21 }
			30 = { country_event = kyo_seija_distribution.22 }
		}
	}
}
