# 只抵扣常驻的 100；200 师时不压制，201 师起合计为 0。
Renko_LUN_suppress_infantry_above_200 = {
    allowed = { original_tag = LUN }
    enable = {
        has_army_size = {
            size > 200
            type = infantry
        }
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = role_ratio
        id = Renko_LUN_infantry
        value = -100
    }
}
