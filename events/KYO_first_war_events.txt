# 事件內容概要：最後通牒。日本要求撤出東京、交出參與者並封閉裂口；百妖議會表決後正式宣戰。
country_event = {
	id = KYO_first_war.2
	title = KYO_first_war.2.t
	desc = KYO_first_war.2.d

	picture = GFX_report_event_first_war_2
	is_triggered_only = yes

	option = {
		name = KYO_first_war.2.a
		ai_chance = {
			base = 100
		}

		if = {
			limit = {
				JAP = {
					exists = yes
				}
			}

			if = {
				limit = {
					NOT = {
						has_war_with = JAP
					}
				}
				declare_war_on = {
					target = JAP
					type = annex_everything
				}
			}

			
			activate_mission = KYO_tokyo_bridgehead_mission
			country_event = {
				id = KYO_first_war.3
				days = 15
			}
			news_event = {
				id = kyo_news.2
				days = 1
			}
			country_event = {
				id = KYO_first_war.20
				days = 1
			}
		}
		else = {
			set_country_flag = KYO_first_war_skipped_japan_absent
			add_political_power = 50
		}
	}
}
